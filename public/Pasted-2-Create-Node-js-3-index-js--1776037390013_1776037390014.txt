const express = require("express");
const admin = require("firebase-admin");
const { RouterOSClient } = require("routeros-client");

const app = express();
app.use(express.json());

// 🔥 Firebase
const serviceAccount = require("./serviceAccount.json");

admin.initializeApp({
  credential: admin.credential.cert(serviceAccount),
});

const db = admin.firestore();

app.post("/addCard", async (req, res) => {
  const { userCode, cardNumber } = req.body;

  if (!userCode || !cardNumber) {
    return res.json({ error: "❌ بيانات ناقصة" });
  }

  try {
    // 🔍 تحقق هل الكرت مستخدم
    const usedRef = db.collection("usedCards").doc(cardNumber);
    const usedDoc = await usedRef.get();

    if (usedDoc.exists) {
      return res.json({ error: "❌ الكرت مستخدم مسبقاً" });
    }

    // 🔌 MikroTik
    const client = new RouterOSClient({
      host: "192.168.88.1",
      user: "admin",
      password: "123456",
    });

    const conn = await client.connect();

    const users = await conn.menu("/ip/hotspot/user").where({
      name: cardNumber,
    }).get();

    if (users.length === 0) {
      return res.json({ error: "❌ الكرت غير موجود" });
    }

    // 🎯 نقاط
    const pointsToAdd = 10;

    const userRef = db.collection("users").doc(userCode);
    const userDoc = await userRef.get();

    const newPoints = (userDoc.data().points || 0) + pointsToAdd;

    await userRef.update({
      points: newPoints,
      logs: admin.firestore.FieldValue.arrayUnion(
        `💳 ${cardNumber} +${pointsToAdd}`
      ),
    });

    await usedRef.set({
      usedBy: userCode,
      date: new Date(),
    });

    res.json({
      success: true,
      newPoints,
      pointsAdded: pointsToAdd,
    });

  } catch (e) {
    console.error(e);
    res.json({ error: "❌ خطأ في السيرفر" });
  }
});

app.listen(3000, () => console.log("Server running"));
